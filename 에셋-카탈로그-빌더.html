CATALOG 등록된 에셋

MANIFEST JSON 내보내기 / 가져오기

내보낸 JSON은 브라우저 저장소에도 자동 보존돼요(localStorage) — 새로고침해도 유지됩니다. 앱은 이 매니페스트를 loadCatalog(json)으로 파싱합니다.

매니페스트 스키마 (앱 계약):

// 앱이 loadCatalog()로 읽는 계약 (src/assets/assetCatalog.ts) { "schema": "armakeup.assetCatalog", "version": 1, "generatedAt": "<ISO 8601>", // 선택 "entries": [ { "id": "<유일 식별자>", "kind": "mask" | "colorArt" | "textureMap", "region": "<kind별 유효 부위>", // mask: blush|highlighter|contour|eyeshadow // colorArt: brow|eyeliner|lip|blush|aegyo // textureMap: lip|eyeshadow|blush "name": "<표시 이름>", "tags": ["…"], "version": 1, // 에셋 리비전(양수) "uri": "<앱이 로드할 경로/URI>", "thumbnail": "<dataURL 또는 uri>", // 선택 "note": "<메모>", // 선택 "createdAt": 1720000000000 // 선택, epoch ms } ] }
